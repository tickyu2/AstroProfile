def calculate_element_distribution(natal_chart):
    """
    Calculate element distribution from natal chart
    Returns dominant element and percentages
    """
    
    elements = {
        'Fire': 0,
        'Earth': 0,
        'Air': 0,
        'Water': 0
    }
    
    # Element mapping
    element_map = {
        'Aries': 'Fire', 'Leo': 'Fire', 'Sagittarius': 'Fire',
        'Taurus': 'Earth', 'Virgo': 'Earth', 'Capricorn': 'Earth',
        'Gemini': 'Air', 'Libra': 'Air', 'Aquarius': 'Air',
        'Cancer': 'Water', 'Scorpio': 'Water', 'Pisces': 'Water'
    }
    
    # Count the Big 8
    placements = [
        'Sun', 'Moon', 'Mercury', 'Venus',
        'Mars', 'Jupiter', 'Saturn', 'Ascendant'
    ]
    
    for placement in placements:
        sign = natal_chart[placement].sign
        element = element_map[sign]
        elements[element] += 1
    
    # Calculate percentages
    total = 8  # Always 8 points
    percentages = {
        element: (count / total) * 100
        for element, count in elements.items()
    }
    
    # Determine dominant (4+ = 50%+)
    dominant = max(elements, key=elements.get)
    is_dominant = elements[dominant] >= 4
    
    return {
        'counts': elements,
        'percentages': percentages,
        'dominant': dominant,
        'is_dominant': is_dominant,
        'test_to_use': get_test_name(dominant) if is_dominant else None
    }

def get_test_name(element):
    """Map element to test name"""
    tests = {
        'Fire': 'campfire',
        'Earth': 'garden',
        'Water': 'great_lakes',
        'Air': 'kite'
    }
    return tests[element]
