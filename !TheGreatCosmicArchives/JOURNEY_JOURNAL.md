### **Encryption**

```javascript
// Client-side encryption
import CryptoJS from 'crypto-js'

function encryptEntry(entryText, userKey) {
  return CryptoJS.AES.encrypt(entryText, userKey).toString()
}

function decryptEntry(encryptedText, userKey) {
  const bytes = CryptoJS.AES.decrypt(encryptedText, userKey)
  return bytes.toString(CryptoJS.enc.Utf8)
}

// User's key derived from password
// Never sent to server
// Only they can read their journal
```
