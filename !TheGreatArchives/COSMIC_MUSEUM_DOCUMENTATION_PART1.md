        Check if this is known threat or variant
        Like immune system recognizing flu variants

        # Exact match?
        if threat_signature in self.threat_database:
            return {
                'known': True,
                'signature': threat_signature,
                'encounters': self.threat_database[threat_signature]['encounter_count'],
                'recommended_response': get_proven_defense(threat_signature)
            }
        
        # Variant of known threat?
        for known_signature, threat_data in self.threat_database.items():
            similarity = calculate_similarity(threat_signature, known_signature)
            
            if similarity > 0.85:  # 85% similar
                # It's a variant!
                return {

                    'is_variant': True,
                    'parent_threat': known_signature,
                    'similarity': similarity,
                    'mutation_detected': True,
                    'recommended_response': adapt_defense_for_variant(
                        threat_data['defense_strategy'],
                        similarity
                    )
                }

            'known': False,
            'is_variant': False,
            'novel_threat': True,
