```python
from pqcrypto.kem import kyber1024

# Alice generates keypair
alice_public, alice_private = kyber1024.generate_keypair()

# Bob creates shared secret
ciphertext, shared_secret_bob = kyber1024.encapsulate(alice_public)

# Alice derives same shared secret
shared_secret_alice = kyber1024.decapsulate(
    ciphertext=ciphertext,
    private_key=alice_private
)

# shared_secret_alice == shared_secret_bob
# Now they can communicate with symmetric encryption
# Quantum computers can't break this
```
