import React, { useState } from 'react'
import { motion, AnimatePresence } from 'framer-motion'
import CenterMedallion from './CenterMedallion'
import PortalButton from './PortalButton'
import SixSoulQuestions from './SixSoulQuestions'
import CompatibilityDiscovery from './CompatibilityDiscovery'
import { getSoulQuestions } from '../../utils/mbti/mbtiSoulQuestions'
import { getTopCompatibleTypes } from '../../utils/mbti/mbtiCompatibilityEngine'

export default function MBTIRoseWindow({ userType, profile }) {
    const [activeLayer, setActiveLayer] = useState(null) // null | 'soul' | 'compatibility'
    
    if (!userType) {
        return <EmptyState profile={profile} />
    }
    
    const soulData = getSoulQuestions(userType)
    const topMatches = getTopCompatibleTypes(userType, 8)
    
    return (
        <motion.div 
            className="rose-window-container"
            initial={{ opacity: 0, scale: 0.95 }}
            animate={{ opacity: 1, scale: 1 }}
            transition={{ duration: 0.8, ease: "easeOut" }}
        >
            
            {/* Background: Dark gradient with stars */}
            <div className="absolute inset-0 bg-gradient-to-br from-slate-900 via-purple-950 to-slate-900">
                <div className="stars-background" />
            </div>
            
            {/* Center Medallion: User's Soul */}
            <CenterMedallion 
                type={userType}
                profile={profile}
            />
            
            {/* Portal Buttons: Two Exploration Paths */}
            <div className="portal-buttons-container">
                
                <PortalButton
                    icon="🔍"
                    title="DISCOVER YOUR SOUL"
                    subtitle="Six sacred questions reveal your essence"
                    color="orange"
                    isActive={activeLayer === 'soul'}
                    onClick={() => setActiveLayer(activeLayer === 'soul' ? null : 'soul')}
                    delay={0.3}
                />
                
                <PortalButton
                    icon="💜"
                    title="EXPLORE COMPATIBILITY"
                    subtitle="Find your Golden Pair among the cosmos"
                    color="purple"
                    isActive={activeLayer === 'compatibility'}
                    onClick={() => setActiveLayer(activeLayer === 'compatibility' ? null : 'compatibility')}
                    delay={0.6}
                />
                
            </div>
            
            {/* Expanded Layers: Rose Petals or Compatibility Circle */}
            <AnimatePresence mode="wait">
                
                {activeLayer === 'soul' && (
                    <motion.div
                        key="soul-layer"
                        initial={{ opacity: 0, scale: 0.8, rotate: -180 }}
                        animate={{ opacity: 1, scale: 1, rotate: 0 }}
                        exit={{ opacity: 0, scale: 0.8, rotate: 180 }}
                        transition={{ 
                            type: "spring",
                            stiffness: 100,
                            damping: 20
                        }}
                    >
                        <SixSoulQuestions 
                            type={userType}
                            soulData={soulData}
                        />
                    </motion.div>
                )}
                
                {activeLayer === 'compatibility' && (
                    <motion.div
                        key="compatibility-layer"
                        initial={{ opacity: 0, scale: 0.8 }}
                        animate={{ opacity: 1, scale: 1 }}
                        exit={{ opacity: 0, scale: 0.8 }}
                        transition={{ 
                            type: "spring",
                            stiffness: 100,
                            damping: 20
                        }}
                    >

                            userType={userType}

                        />

import React from 'react'
import { motion } from 'framer-motion'

export default function PortalButton({ 
    icon, 
    title, 
    subtitle, 
    color, 
    isActive, 
    onClick,
    delay = 0 
}) {
    
    const colorClasses = {
        orange: {
            gradient: 'from-orange-500 via-amber-500 to-orange-500',
            glow: 'rgba(249, 115, 22, 0.6)',
            glowHover: 'rgba(249, 115, 22, 0.9)'
        },
        purple: {
            gradient: 'from-purple-600 via-violet-500 to-purple-600',
            glow: 'rgba(124, 58, 237, 0.6)',
            glowHover: 'rgba(124, 58, 237, 0.9)'
        }
    }
    
    const colors = colorClasses[color]
    
    return (
        <motion.button
            className="portal-button"
            initial={{ y: 100, opacity: 0 }}
            animate={{ y: 0, opacity: 1 }}
            transition={{
                delay,
                type: "spring",
                stiffness: 100,
                damping: 20
            }}
            whileHover={{ 
                scale: 1.02,
                boxShadow: `0 0 60px ${colors.glowHover}`
            }}
            whileTap={{ scale: 0.98 }}
            onClick={onClick}
            style={{
                background: `linear-gradient(135deg, ${colors.gradient})`,
                boxShadow: isActive 
                    ? `0 0 40px ${colors.glowHover}, inset 0 0 20px rgba(255,255,255,0.2)`
                    : `0 0 20px ${colors.glow}`
            }}
        >
            
            {/* Glass layers */}
            <div className="glass-layer-1" />
            <div className="glass-layer-2" />
            <div className="glass-layer-3" />
            
            {/* Light rays */}
            <div className="light-rays">
                <motion.div 
                    className="ray"
                    animate={{
                        opacity: [0.2, 0.6, 0.2],
                        scaleY: [0.8, 1.2, 0.8]
                    }}
                    transition={{
                        duration: 2,
                        repeat: Infinity,
                        ease: "easeInOut"
                    }}
                />
            </div>
            
            {/* Content */}
            <div className="portal-content">
                <span className="portal-icon">{icon}</span>
                <h3 className="portal-title">{title}</h3>
                <p className="portal-subtitle">{subtitle}</p>
            </div>
            
            {/* Active indicator */}
            {isActive && (
                <motion.div
                    className="active-indicator"
                    initial={{ scale: 0 }}
                    animate={{ scale: 1 }}
                    transition={{ type: "spring" }}
                >
                    ✓
                </motion.div>
            )}
            
        </motion.button>
    )
}
