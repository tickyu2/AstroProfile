### Core calculation function:

```javascript
const swisseph = require('swisseph');

export const calculateBasicChart = functions.https.onCall(async (data) => {
  const { birthDate, birthTime, latitude, longitude } = data;
  
  // Initialize Swiss Ephemeris
  swisseph.swe_set_ephe_path('./ephemeris'); // Data files location
  
  // Calculate Julian Day
  const julianDay = swisseph.swe_julday(
    birthDate.year, birthDate.month, birthDate.day,
    birthTime.hour + (birthTime.minute / 60), 
    swisseph.SE_GREG_CAL
  );
  
  // Get Sun position
  const sun = swisseph.swe_calc_ut(julianDay, swisseph.SE_SUN, swisseph.SEFLG_SWIEPH);
  
  // Get Moon position  
  const moon = swisseph.swe_calc_ut(julianDay, swisseph.SE_MOON, swisseph.SEFLG_SWIEPH);
  
  // Calculate Rising sign (Ascendant)
  const houses = swisseph.swe_houses(julianDay, latitude, longitude, 'P'); // Placidus
  
  return {
    sun: { longitude: sun.longitude, sign: getZodiacSign(sun.longitude) },
    moon: { longitude: moon.longitude, sign: getZodiacSign(moon.longitude) },
    rising: { longitude: houses.ascendant, sign: getZodiacSign(houses.ascendant) }
  };
});
```
