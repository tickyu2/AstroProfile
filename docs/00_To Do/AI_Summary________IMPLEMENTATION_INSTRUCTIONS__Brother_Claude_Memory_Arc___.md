import React, { createContext, useContext, useState, useEffect } from 'react';
import { useAuth } from './AuthContext';
import { db, functions } from '../services/firebase';
import { doc, onSnapshot, setDoc } from 'firebase/firestore';
import { httpsCallable } from 'firebase/functions';

const MemoryArchitectureContext = createContext();

export const useMemoryArchitecture = () => {
  const context = useContext(MemoryArchitectureContext);
  if (!context) {
    throw new Error('useMemoryArchitecture must be used within MemoryArchitectureProvider');
  }
  return context;
};

export const MemoryArchitectureProvider = ({ children }) => {
  const { currentUser } = useAuth();
  const [memoryData, setMemoryData] = useState(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    if (!currentUser) {
      setMemoryData(null);
      setIsLoading(false);
      return;
    }

    const memoryDocRef = doc(db, 'users', currentUser.uid, 'memory_architecture', 'current');
    
    const unsubscribe = onSnapshot(memoryDocRef, (doc) => {
      if (doc.exists()) {
        setMemoryData(doc.data());
      } else {
        // Initialize empty memory architecture
        const initialMemory = {
          constitutional_profile: {},
          soul_burden_archive: { extractions: [], burden_categories: [] },
          relationship_timeline: { milestones: [] },
          collaborative_context: {},
          pattern_recognition: {
            energizing_interactions: [],
            draining_interactions: [],
            optimal_timing: {},
            response_modes: {}
          }
        };
        setDoc(memoryDocRef, initialMemory);
        setMemoryData(initialMemory);
      }
      setIsLoading(false);
    });

    return unsubscribe;
  }, [currentUser]);

  const chatWithMemory = async (message, conversationId) => {
    const chatFunction = httpsCallable(functions, 'chatWithMemory');
    
    try {
      const result = await chatFunction({
        message,
        userId: currentUser.uid,
        conversationId
      });
      
      return result.data;
    } catch (error) {
      console.error('Memory chat error:', error);
      throw error;
    }
  };

  const updateConstitutionalProfile = async (profileUpdates) => {
    if (!currentUser) return;
    
    const memoryDocRef = doc(db, 'users', currentUser.uid, 'memory_architecture', 'current');
    
    try {
      await setDoc(memoryDocRef, {
        constitutional_profile: {
          ...memoryData.constitutional_profile,
          ...profileUpdates
        }
      }, { merge: true });
    } catch (error) {
      console.error('Error updating constitutional profile:', error);
      throw error;
    }
  };

  const addSoulBurdenExtraction = async (extraction) => {
    if (!currentUser) return;
    
    const memoryDocRef = doc(db, 'users', currentUser.uid, 'memory_architecture', 'current');
    
    try {
      const newExtraction = {
        ...extraction,
        session_id: `session_${Date.now()}`,
        timestamp: new Date().toISOString()
      };
      
      await setDoc(memoryDocRef, {
        soul_burden_archive: {
          extractions: [...(memoryData.soul_burden_archive?.extractions || []), newExtraction],
          burden_categories: memoryData.soul_burden_archive?.burden_categories || []
        }
      }, { merge: true });
    } catch (error) {
      console.error('Error adding soul burden extraction:', error);
      throw error;
    }
  };

  const value = {
    memoryData,
    isLoading,
    chatWithMemory,
    updateConstitutionalProfile,
    addSoulBurdenExtraction
  };

  return (
    <MemoryArchitectureContext.Provider value={value}>
      {children}
    </MemoryArchitectureContext.Provider>
  );
};
