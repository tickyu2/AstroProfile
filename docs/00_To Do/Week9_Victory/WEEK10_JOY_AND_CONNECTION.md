/**
 * Inside Jokes Engine
 * Detects, stores, and uses personalized humor
 */

class InsideJokesEngine {
  
  constructor() {
    // Joke detection patterns
    this.patterns = {
      creativePhrase: /(".*?")|('.*?')/g,  // Quoted phrases
      unusualWords: ['bonkers', 'shenanigans', 'catastrophe', 'disaster'],
      emoji: /[😀-🙏]/g,  // Detect emoji usage
      laughReactions: ['haha', 'lol', '😂', '😄', 'that's funny', 'love it']
    };
    
    // Joke categories
    this.categories = {
      NICKNAME: 'nickname',       // Pet names for things
      PHRASE: 'creative_phrase',  // Original expressions
      MOMENT: 'memorable_moment', // Shared experiences
      REFERENCE: 'shared_reference', // Movies, songs, etc.
      CHARACTER: 'recurring_character'  // People/pets in user's life
    };
  }
  
  /**
   * Detect potential inside jokes from conversation
   */
  async detectPotentialJoke(userId, message, context) {
    const candidates = [];
    
    // Check for creative phrases in quotes
    const quotedPhrases = message.match(this.patterns.creativePhrase);
    if (quotedPhrases) {
      quotedPhrases.forEach(phrase => {
        candidates.push({
          type: this.categories.PHRASE,
          content: phrase.replace(/['"]/g, ''),
          reason: 'user_quoted',
          confidence: 0.8
        });
      });
    }
    
    // Check for unusual/funny word combinations
    const words = message.toLowerCase().split(' ');
    const unusualWords = words.filter(w => 
      this.patterns.unusualWords.includes(w)
    );
    
    if (unusualWords.length > 0) {
      candidates.push({
        type: this.categories.PHRASE,
        content: message,
        reason: 'unusual_vocabulary',
        confidence: 0.6
      });
    }
    
    // Check for laugh reactions (indicates humor)
    const hasLaughReaction = this.patterns.laughReactions.some(r => 
      message.toLowerCase().includes(r)
    );
    
    if (hasLaughReaction && context.previousMessage) {
      // Previous message might be the joke
      candidates.push({
        type: this.categories.MOMENT,
        content: context.previousMessage,
        reason: 'user_laughed',
        confidence: 0.9
      });
    }
    
    // Check for recurring characters (names with context)
    const properNouns = this.extractProperNouns(message);
    if (properNouns.length > 0) {
      properNouns.forEach(noun => {
        candidates.push({
          type: this.categories.CHARACTER,
          content: noun,
          reason: 'recurring_character',
          confidence: 0.7
        });
      });
    }
    
    // Store high-confidence candidates
    for (const candidate of candidates) {
      if (candidate.confidence >= 0.7) {
        await this.storeJokeCandidate(userId, candidate, context);
      }
    }
    
    return candidates;
  }
  
  /**
   * Store inside joke candidate
   */
  async storeJokeCandidate(userId, jokeData, context) {
    const db = require('../config/genesisDatabase');
    
    await db.query(`
      INSERT INTO luna_inside_jokes (
        user_id, category, content, original_context,
        confidence, status, first_seen
      ) VALUES ($1, $2, $3, $4, $5, 'candidate', NOW())
    `, [
      userId,
      jokeData.type,
      jokeData.content,
      JSON.stringify(context),
      jokeData.confidence
    ]);
  }
  
  /**
   * Confirm joke (user reacted positively when Luna used it)
   */
  async confirmJoke(userId, jokeId) {
    const db = require('../config/genesisDatabase');
    
    await db.query(`
      UPDATE luna_inside_jokes
      SET 
        status = 'confirmed',
        use_count = use_count + 1,
        last_used = NOW()
      WHERE user_id = $1 AND id = $2
    `, [userId, jokeId]);
  }
  
  /**
   * Get active inside jokes for user
   */
  async getActiveJokes(userId, category = null) {
    const db = require('../config/genesisDatabase');
    
    let query = `
      SELECT * FROM luna_inside_jokes
      WHERE user_id = $1 AND status = 'confirmed'
    `;
    
    const params = [userId];
    
    if (category) {
      query += ' AND category = $2';
      params.push(category);
    }
    
    query += ' ORDER BY use_count DESC, last_used DESC';
    
    const result = await db.query(query, params);
    return result.rows;
  }
  
  /**
   * Suggest when to use inside joke
   */
  async suggestJokeUsage(userId, currentContext) {
    const jokes = await this.getActiveJokes(userId);
    
    if (jokes.length === 0) return null;
    
    // Find relevant jokes based on context
    const relevant = jokes.filter(joke => {
      // Check if context matches
      const originalContext = JSON.parse(joke.original_context);
      
      // Similar topics
      if (this.hasOverlap(currentContext.topics, originalContext.topics)) {
        return true;
      }
      
      // Similar emotion
      if (currentContext.emotion === originalContext.emotion) {
        return true;
      }
      
      // Don't overuse - not used recently
      const hoursSinceUse = this.getHoursSince(joke.last_used);
      if (hoursSinceUse < 24) {
        return false;
      }
      
      return false;
    });
    
    if (relevant.length === 0) return null;
    
    // Return highest use_count (most successful joke)
    return relevant.sort((a, b) => b.use_count - a.use_count)[0];
  }
  
  /**
   * Extract proper nouns (potential character names)
   */
  extractProperNouns(text) {
    // Simple capitalized word detection
    // In production, use NER (Named Entity Recognition)
    const words = text.split(' ');
    const properNouns = words.filter(word => 
      word.length > 2 &&
      word[0] === word[0].toUpperCase() &&
      word.slice(1) === word.slice(1).toLowerCase()
    );
    
    return properNouns;
  }
  
  hasOverlap(arr1, arr2) {
    if (!arr1 || !arr2) return false;
    return arr1.some(item => arr2.includes(item));
  }
  
  getHoursSince(timestamp) {
    const now = new Date();
    const then = new Date(timestamp);
    return (now - then) / (1000 * 60 * 60);
  }
}

module.exports = InsideJokesEngine;

      }
    };
  }
  
  /**
   * Generate playful banter response
   */
  async generateBanter(userMessage, context) {
    const { emotion, relationship, recentMessages } = context;
    
    // Only banter if relationship is strong enough
    if (relationship.affection < 6) {
      return null;  // Not close enough yet
    }
    
    // Match banter style to situation
    let banterStyle = this.selectBanterStyle(userMessage, emotion);
    
    if (!banterStyle) return null;
    
    // Generate banter
    const banter = this.generateFromStyle(banterStyle, userMessage, context);
    
    return {
      text: banter,
      style: banterStyle,
      shouldUse: this.shouldUseBanter(context)
    };
  }
  
  /**
   * Select appropriate banter style
   */
  selectBanterStyle(message, emotion) {
    const lower = message.toLowerCase();
    
    // Affectionate teasing (for confidence or mild complaints)
    if (lower.includes("i'm the best") || 
        lower.includes("i'm amazing") ||
        lower.includes("i'm so smart")) {
      return 'affectionateTeasing';
    }
    
    // Absurd escalation (for dramatic statements)
    if (lower.includes("so tired") ||
        lower.includes("so stressed") ||
        lower.includes("the worst")) {
      return 'absurdEscalation';
    }
    
    // Clever observation (for complaints about things)
    if (lower.includes("why do") ||
        lower.includes("always") ||
        lower.includes("never")) {
      return 'cleverObservation';
    }
    
    return null;
  }
  
  /**
   * Generate banter from style template
   */
  generateFromStyle(style, message, context) {
    // This would use LLM with banter-specific prompting
    // Placeholder for example
    
    const examples = {
      affectionateTeasing: "Oh absolutely. The humility alone proves it 😏",
      absurdEscalation: "You've transcended tiredness. You're in the mystical realm of 'why are my eyes doing this'",
      cleverObservation: "Keys are sentient beings with a vendetta. It's science."
    };
    
    return examples[style];
  }
  
  /**
   * Should we use banter right now?
   */
  shouldUseBanter(context) {
    // Don't banter if user is genuinely distressed
    if (context.emotion.primary === 'sadness' && context.emotion.intensity > 7) {
      return false;
    }
    
    // Don't banter in serious contexts
    if (context.contextTags?.includes('grief') || 
        context.contextTags?.includes('trauma')) {
      return false;
    }
    
    // Good to banter!
    return true;
  }
  
  /**
   * Get appropriate pun
   */
  getPun(context) {
    if (context.mentions?.includes('cheese')) {
      return "Having a 'grate' day! 🧀";
    }
    
    if (context.emotion === 'worried') {
      return "Donut worry about it! 🍩";
    }
    
    if (context.occasion === 'compliment') {
      return "You're one in a melon! 🍉";
    }
    
    return null;
  }
}

module.exports = BanterSystem;
