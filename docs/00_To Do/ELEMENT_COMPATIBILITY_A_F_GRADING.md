function normalizeCompatibilityScores(
  rawScores: Record<string, number>
): Record<string, number> {
  
  const scores = Object.values(rawScores);
  const maxScore = Math.max(...scores);
  const minScore = Math.min(...scores);
  const range = maxScore - minScore;
  
  const normalized: Record<string, number> = {};
  
  for (const [cusp, rawScore] of Object.entries(rawScores)) {
    // Scale to 25-95 range
    // Best match = 95%, Worst = 25%
    const normalizedScore = ((rawScore - minScore) / range) × 70 + 25;
    normalized[cusp] = Math.round(normalizedScore * 10) / 10;
  }
  
  return normalized;
}

function calculateElementCompatibility(
  user: ElementProfile,
  partner: ElementProfile
): CompatibilityScore {
  
  // STEP 1: Calculate raw components
  const overlap = calculateOverlap(user, partner);
  const complement = calculateCriticalComplement(user, partner);
  const communication = calculatePartnerCommunication(partner);
  
  // STEP 2: Weighted sum (20/60/20)
  const rawScore = (
    overlap * 0.20 +
    complement * 0.60 +
    communication * 0.20
  );
  
  // STEP 3: Store raw score for later normalization
  return {
    rawScore,
    overlap,
    complement,
    communication
  };
}

function calculateCriticalComplement(
  user: ElementProfile,
  partner: ElementProfile
): number {
  
  // Identify CRITICAL deficits (elements < 20%)
  const userCriticalDeficits = {};
  const userMinorDeficits = {};
  
  for (const element of ['fire', 'earth', 'air', 'water']) {
    const value = user.elements[element];
    if (value < 15) {
      // Severe deficit
      userCriticalDeficits[element] = 20 - value;
    } else if (value < 20) {
      // Minor deficit
      userMinorDeficits[element] = 20 - value;
    }
  }
  
  // Calculate what partner fills
  const criticalTotal = Object.values(userCriticalDeficits).reduce((a, b) => a + b, 0);
  const minorTotal = Object.values(userMinorDeficits).reduce((a, b) => a + b, 0);
  
  let criticalFilled = 0;
  for (const [element, deficit] of Object.entries(userCriticalDeficits)) {
    const partnerStrength = Math.max(0, partner.elements[element] - 20);
    criticalFilled += Math.min(deficit, partnerStrength);
  }
  
  let minorFilled = 0;
  for (const [element, deficit] of Object.entries(userMinorDeficits)) {
    const partnerStrength = Math.max(0, partner.elements[element] - 20);
    minorFilled += Math.min(deficit, partnerStrength);
  }
  
  // Critical deficits are TWICE as important
  const criticalRate = criticalTotal > 0 ? (criticalFilled / criticalTotal) : 1;
  const minorRate = minorTotal > 0 ? (minorFilled / minorTotal) : 1;
  
  // Weight: 70% critical, 30% minor
  const userReception = criticalRate * 0.70 + minorRate * 0.30;
  
  // Also calculate partner reception (but weight less)
  const partnerDeficits = calculateAllDeficits(partner);
  const partnerDeficitTotal = Object.values(partnerDeficits).reduce((a, b) => a + b, 0);
  
  let partnerFilled = 0;
  for (const [element, deficit] of Object.entries(partnerDeficits)) {
    const userStrength = Math.max(0, user.elements[element] - 20);
    partnerFilled += Math.min(deficit, userStrength);
  }
  
  const partnerReception = partnerDeficitTotal > 0 
    ? (partnerFilled / partnerDeficitTotal) 
    : 1;
  
  // User is searching, so weight their reception more
  return (userReception * 0.70 + partnerReception * 0.30) * 100;
}

function calculatePartnerCommunication(partner: ElementProfile): number {
  // Partner's communication capacity (NOT limited by user)
  return ((partner.elements.air + partner.elements.water) / 2);
}

// After calculating ALL cusps:
function normalizeAllScores(
  results: Record<string, CompatibilityScore>
): Record<string, number> {
  
  const rawScores = Object.entries(results).map(([cusp, score]) => ({
    cusp,
    raw: score.rawScore
  }));
  
  const maxRaw = Math.max(...rawScores.map(s => s.raw));
  const minRaw = Math.min(...rawScores.map(s => s.raw));
  const range = maxRaw - minRaw;
  
  const normalized: Record<string, number> = {};
  
  for (const { cusp, raw } of rawScores) {
    // Scale to 25-95 range
    normalized[cusp] = ((raw - minRaw) / range) * 70 + 25;
  }
  
  return normalized;
}
