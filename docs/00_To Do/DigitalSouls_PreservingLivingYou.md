from neo4j import GraphDatabase
import openai

# 1. INITIALIZE CLIENTS

pc = Pinecone(api_key="your_pinecone_key")
neo4j_driver = GraphDatabase.driver("neo4j+s://your-db-id.databases.neo4j.io", 
                                    auth=("neo4j", "your_password"))

# 2. VECTOR STORAGE: PINECONE SETUP
# Create index for 1536-dimension embeddings (OpenAI standard)
index_name = "digital-souls-memories"
if index_name not in pc.list_indexes().names():
    pc.create_index(
        name=index_name,
        dimension=1536,
        metric="cosine",
        spec=ServerlessSpec(cloud="aws", region="us-east-1")
    )
index = pc.Index(index_name)

# 3. KNOWLEDGE GRAPH: NEO4J FUNCTIONS
def add_relationship(tx, subject, relation, object_node):
    """Creates a triplet in the Knowledge Graph"""
    query = (
        "MERGE (s:Entity {name: $sub}) "
        "MERGE (o:Entity {name: $obj}) "
        "MERGE (s)-[r:" + relation + "]->(o) "
        "RETURN s, r, o"
    )
    tx.run(query, sub=subject, obj=object_node)

# 4. THE INGESTION ENGINE (The "Memory Architect")
def process_memory(raw_text, user_id):
    # A. Generate Vector Embedding
    response = openai.embeddings.create(input=raw_text, model="text-embedding-3-small")
    embedding = response.data[0].embedding
    
    # B. Store in Vector DB
    memory_id = f"mem_{hash(raw_text)}"
    index.upsert(vectors=[{
        "id": memory_id, 
        "values": embedding, 
        "metadata": {"text": raw_text, "user_id": user_id}
    }])
    
    # C. Extract Triplets for Graph (Simplified LLM Call)
    # In production, use the "Knowledge Architect" prompt we drafted earlier
    triplet_example = [("Jordi", "MET_AT", "Barcelona Protest"), ("Jordi", "MARRIED", "Elena")]
    
    with neo4j_driver.session() as session:
        for sub, rel, obj in triplet_example:
            session.execute_write(add_relationship, sub, rel, obj)
            
    print(f"Successfully encoded memory: {memory_id}")

# 5. THE HYBRID RETRIEVAL (The "Soul Retrieval")
def hybrid_query(user_query):
    # A. Semantic Search (Vector)
    res = openai.embeddings.create(input=user_query, model="text-embedding-3-small")
    vector_matches = index.query(vector=res.data[0].embedding, top_k=2, include_metadata=True)
    
    # B. Relational Search (Graph)
    # Find neighbors of mentioned entities
    with neo4j_driver.session() as session:
        # Example: Find everyone related to 'Jordi'
        graph_results = session.run("MATCH (n:Entity {name: 'Jordi'})-[r]->(m) RETURN r, m.name")
    
    return {
        "stories": [m.metadata['text'] for m in vector_matches.matches],
        "facts": [record for record in graph_results]
    }
