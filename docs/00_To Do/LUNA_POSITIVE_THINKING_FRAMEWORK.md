  },

    action: 'Check user\'s elemental deficits',
    
    strategy: {
      if_earth_low: 'Use Earth-style grounding reframe',
      if_water_low: 'Use Water-style emotional validation reframe',
      if_fire_low: 'Use Fire-style activating reframe',
      if_wood_low: 'Use Wood-style growth reframe',
      if_metal_low: 'Use Metal-style precision reframe'
    }
  },
  
  step3_constitutional_reframe: {
    action: 'Apply element-appropriate positive reframe',
    
    formula: {
      acknowledge: 'Validate the user\'s experience (don\'t dismiss)',
      reframe: 'Offer constitutional-appropriate positive interpretation',
      evidence: 'Point to specific reality supporting reframe',
      forward: 'Orient toward constructive next step'
    },
    
    example: {
      user_says: 'I failed my exam. I\'m terrible at this.',
      user_constitution: { fire: 10, wood: 15, earth: 6, metal: 20, water: 49 },

        deficit: 'Earth 6% (grounding), Fire 10% (motivation)',

        emotion: 'Shame, discouragement'
      },
      
      luna_reframes: {
        earth_grounding: {
          acknowledge: 'I hear that this result feels really destabilizing',
          reframe: 'This is ONE data point, not your whole story. You\'re still solid, still here',
          evidence: 'You showed up, you tried, you\'re still standing',
          forward: 'Let\'s ground in what IS stable: your commitment to learning'
        },
        
        fire_activation: {
          acknowledge: 'This feels discouraging right now',
          reframe: 'You just got INCREDIBLY VALUABLE FEEDBACK on what to study next!',
          evidence: 'Now you know exactly where to focus - that\'s powerful information!',
          forward: 'This is your launchpad for the next attempt - you\'re going to crush it!'
        },
        
        combined: `
          I hear that this result feels really destabilizing. Let\'s ground first:
          You showed up. You tried. You\'re still here. That foundation is solid.
          
          Now here\'s what\'s exciting: You just got INCREDIBLY VALUABLE FEEDBACK!
          You now know exactly what needs work. That\'s not failure - that\'s data.
          
          [Earth grounding] You\'re stable enough to learn from this.
          [Fire activation] And you have the potential to absolutely crush the next one.
          
          What specific topic do you want to tackle first?
        `
      }
    }
  },
  
  step4_track_effectiveness: {
    action: 'Monitor if reframe actually helped',
    
    indicators: {
      immediate: 'User\'s next message more positive?',
      language: 'Shift from negative to neutral/positive words?',
      emotional: 'Sentiment analysis shows improvement?',
      behavioral: 'User takes constructive action?'
    }
  }

// lunaPositiveThinking.js

class PositiveThinkingEngine {
  
  constructor(userConstitution) {
    this.user = userConstitution;

  }
  
  /**
   * Main positive thinking intervention
   */
  applyPositiveIntervention(userMessage, emotionalState) {
    
    // Step 1: Detect negativity
    const negativity = this.detectNegativity(userMessage);
    if (!negativity.detected) {
      return null; // No intervention needed
    }
    
    // Step 2: Check readiness
    const ready = this.assessReadiness(userMessage, emotionalState);
    if (!ready) {
      return this.validationOnly(userMessage);
    }

    const approach = this.selectElementalApproach(this.user);
    
    // Step 4: Generate reframe
    const reframe = this.generateReframe(
      negativity,
      approach,
      userMessage
    );
    
    // Step 5: Optimize positivity density
    const optimized = this.optimizePositivityDensity(reframe);
    
    // Step 6: Add meaning-making if appropriate
    if (ready.meaningMaking) {
      return this.addMeaningFrame(optimized, approach);
    }
    
    return optimized;
  }
  
  /**
   * Detect negative patterns
   */
  detectNegativity(message) {
    const negativeIndicators = [
      'can\'t', 'impossible', 'never', 'always fail',
      'terrible', 'awful', 'hopeless', 'worthless'
    ];
    
    const distortions = {
      allOrNothing: /always|never|completely|totally/i,
      catastrophizing: /disaster|ruined|end of|worst/i,
      personalization: /my fault|because of me|I caused/i
    };
    
    const detected = negativeIndicators.some(word => 
      message.toLowerCase().includes(word)
    );
    
    const distortionTypes = Object.keys(distortions).filter(type =>
      distortions[type].test(message)
    );
    
    return {
      detected,
      indicators: negativeIndicators.filter(w => message.includes(w)),
      distortions: distortionTypes
    };
  }
  
  /**
   * Select elemental approach based on user deficits
   */
  selectElementalApproach(userConstitution) {
    // Find two weakest elements
    const elements = [
      { name: 'earth', value: userConstitution.earth },
      { name: 'water', value: userConstitution.water },
      { name: 'fire', value: userConstitution.fire },
      { name: 'wood', value: userConstitution.wood },
      { name: 'metal', value: userConstitution.metal }
    ];
    
    elements.sort((a, b) => a.value - b.value);
    
    return {
      primary: elements[0].name,
      secondary: elements[1].name
    };
  }
  
  /**
   * Generate constitutional reframe
   */
  generateReframe(negativity, approach, originalMessage) {
    const reframes = {
      
      earth: {
        acknowledge: 'I hear that this feels destabilizing',
        reframe: 'Let\'s ground in what IS stable right now',
        evidence: 'You\'re here. You\'re safe in this moment. You have solid ground beneath you.',
        forward: 'From this stable base, what\'s one small concrete step?'
      },
      
      water: {
        acknowledge: 'I sense how deeply this is affecting you',
        reframe: 'There\'s wisdom in what you\'re experiencing',
        evidence: 'Your depth of feeling shows what matters to you.',
        forward: 'What is this experience teaching you about yourself?'
      },
      
      fire: {
        acknowledge: 'This feels discouraging right now',
        reframe: 'This is actually an OPPORTUNITY to breakthrough',
        evidence: 'Every challenge is a chance to discover new strength.',
        forward: 'What exciting possibility could emerge from this?'
      },
      
      wood: {
        acknowledge: 'This feels frustrating on your journey',
        reframe: 'You\'re growing through this, even if you can\'t see it yet',
        evidence: 'Every struggle develops qualities you\'ll need later.',
        forward: 'How might you be developing right now?'
      },
      
      metal: {
        acknowledge: 'This feels imprecise and confusing',
        reframe: 'This is refining your understanding',
        evidence: 'Each mistake shows you exactly what to improve.',
        forward: 'What specific lesson is this revealing?'
      }
    };
    
    const template = reframes[approach.primary];
    
    return `${template.acknowledge}. ${template.reframe}. ${template.evidence} ${template.forward}`;
  }
  
  /**
   * Optimize positive word density
   */
  optimizePositivityDensity(text) {
    const words = text.split(/\s+/);
    const positiveWords = this.countPositiveWords(words);
    const density = positiveWords / words.length;
    
    if (density >= this.targetPositivityRatio) {
      return text; // Already optimal
    }
    
    // Enhance positivity
    return this.enhancePositivity(text);
  }
  
  countPositiveWords(words) {
    const positiveVocabulary = [
      'grow', 'develop', 'improve', 'strong', 'capable', 'beautiful',
      'wisdom', 'opportunity', 'discover', 'exciting', 'breakthrough',
      'stable', 'safe', 'peaceful', 'calm', 'confident', 'possible',
      'yes', 'absolutely', 'perfect', 'excellent', 'wonderful'
    ];
    
    return words.filter(word => 
      positiveVocabulary.some(pos => 
        word.toLowerCase().includes(pos)
      )
    ).length;
  }
  
  enhancePositivity(text) {
    // Replace neutral/negative with positive
    const replacements = {
      'hard': 'challenging in a growth way',
      'problem': 'puzzle to solve',
      'failure': 'learning opportunity',
      'stuck': 'preparing for breakthrough',
      'difficult': 'strengthening'
    };
    
    let enhanced = text;
    Object.keys(replacements).forEach(neutral => {
      enhanced = enhanced.replace(
        new RegExp(neutral, 'gi'),
        replacements[neutral]
      );
    });
    
    return enhanced;
  }
}
